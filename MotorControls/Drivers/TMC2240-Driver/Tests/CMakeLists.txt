cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(tmc2240_host_tests LANGUAGES C)
    enable_testing()
    option(TMC_ENABLE_SANITIZERS "Instrument host tests with ASan and UBSan" OFF)
    option(TMC_ENABLE_COVERAGE "Instrument host tests with LLVM coverage" OFF)
    option(TMC_EXTRA_WARNINGS "Enable additional conversion and API diagnostics" ON)
    if(TMC_EXTRA_WARNINGS)
        add_compile_options(-Wconversion -Wsign-conversion -Wshadow -Wformat=2
                            -Wundef -Wstrict-prototypes)
    endif()
    if(TMC_ENABLE_SANITIZERS)
        add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
        add_link_options(-fsanitize=address,undefined)
    endif()
    if(TMC_ENABLE_COVERAGE)
        add_compile_options(-fprofile-instr-generate -fcoverage-mapping)
        add_link_options(-fprofile-instr-generate)
    endif()
endif()

set(TMC2240_DRIVER "${CMAKE_CURRENT_SOURCE_DIR}/../driver")
function(tmc2240_test target)
    add_executable(${target} ${ARGN})
    set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
    target_include_directories(${target} PRIVATE "${TMC2240_DRIVER}/inc" "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(TMC_ENABLE_SANITIZERS)
        target_compile_options(${target} PRIVATE -fno-sanitize-recover=undefined)
    endif()
    add_test(NAME ${target} COMMAND ${target})
    set_tests_properties(${target} PROPERTIES TIMEOUT 120)
    if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
        set_tests_properties(${target} PROPERTIES
            ENVIRONMENT "LLVM_PROFILE_FILE=${CMAKE_CURRENT_BINARY_DIR}/${target}-%p.profraw")
    endif()
endfunction()

foreach(variant IN ITEMS cache1 cache4 nocache custom external_crc external_crc_nocache external_crc_custom)
    set(target "tmc2240_core_${variant}")
    tmc2240_test(${target} test_core.c model_device.c model_core.c model_cache.c
        test_support.c "${TMC2240_DRIVER}/src/TMC2240.c")
    if(variant STREQUAL "cache1")
        target_compile_definitions(${target} PRIVATE TMC2240_IC_CACHE_COUNT=1)
    elseif(variant MATCHES "nocache")
        target_compile_definitions(${target} PRIVATE TMC2240_CACHE=0)
    elseif(variant MATCHES "custom")
        target_compile_definitions(${target} PRIVATE TMC2240_ENABLE_TMC_CACHE=0)
    endif()
    if(variant MATCHES "external_crc")
        target_compile_definitions(${target} PRIVATE TMC_API_EXTERNAL_CRC_TABLE=1)
        target_sources(${target} PRIVATE external_crc.c)
    endif()
endforeach()

foreach(variant IN ITEMS cache1 cache4 nocache custom)
    set(target "tmc2240_hal_${variant}")
    tmc2240_test(${target} test_hal.c model_device.c model_hal.c model_cache.c
        test_support.c "${TMC2240_DRIVER}/src/TMC2240.c" "${TMC2240_DRIVER}/src/tmc2240_hal.c")
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/fake_hal")
    if(variant STREQUAL "cache1")
        target_compile_definitions(${target} PRIVATE TMC2240_IC_CACHE_COUNT=1)
    elseif(variant STREQUAL "nocache")
        target_compile_definitions(${target} PRIVATE TMC2240_CACHE=0)
    elseif(variant STREQUAL "custom")
        target_compile_definitions(${target} PRIVATE TMC2240_ENABLE_TMC_CACHE=0)
    endif()
endforeach()
